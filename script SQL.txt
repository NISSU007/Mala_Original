-- สร้างตาราง Order
CREATE TABLE "Order" (
    Order_ID INTEGER PRIMARY KEY AUTOINCREMENT,   -- รหัสคำสั่งซื้อ
    Table_ID INTEGER,                             -- รหัสโต๊ะที่ทำการสั่ง (FK เชื่อมตาราง Table)
    Order_Type TEXT,                              -- ประเภทคำสั่งซื้อ: 'ทานที่ร้าน', 'รับกลับบ้าน'
    Total_Price REAL,                             -- ยอดราคารวมทั้งหมดของออเดอร์
    Order_Status TEXT,                            -- สถานะออเดอร์: 'รอยืนยัน', 'กำลังเตรียมอาหาร', 'ปรุงเสร็จสิ้น', 'พร้อมเสิร์ฟ', 'เสร็จสิ้น', 'ยกเลิก'
    Order_Date_Time DATETIME DEFAULT CURRENT_TIMESTAMP, -- วันเวลาที่สั่งซื้อ
    Reference_Code TEXT                           -- รหัสอ้างอิงออเดอร์ (เช่น หมายเลขคิว/เลขบิล)
);

-- สร้างตาราง Table
CREATE TABLE "Table" (
    Table_ID INTEGER PRIMARY KEY AUTOINCREMENT,   -- รหัสโต๊ะ
    Table_Number TEXT,                            -- หมายเลขโต๊ะ (เช่น T-01, T-02)
    Table_Name TEXT,                              -- ชื่อเรียกโต๊ะ
    Table_Status TEXT,                            -- สถานะโต๊ะ: 'พร้อมใช้งาน', 'มีลูกค้าใช้งาน', 'จองแล้ว', 'ปิดปรับปรุง'
    QR_Code_URL TEXT                              -- ลิงก์รูปภาพ QR Code สั่งอาหารประจำโต๊ะ
);

-- สร้างตาราง Menu
CREATE TABLE Menu (
    Menu_ID INTEGER PRIMARY KEY AUTOINCREMENT,    -- รหัสรายการเมนู
    Category_ID INTEGER,                          -- รหัสหมวดหมู่ (FK เชื่อมตาราง Category)
    Menu_Name TEXT,                               -- ชื่อรายการอาหาร
    Price REAL,                                   -- ราคาต่อหน่วย (บาท)
    Image_URL TEXT,                               -- พาธหรือลิงก์รูปภาพเมนู
    Description TEXT,                             -- คำอธิบายเมนูเพิ่มเติม
    Stock_Quantity REAL,                          -- จำนวนสินค้าคงเหลือพร้อมขาย
    Is_Available INTEGER DEFAULT 1                -- สถานะเปิด/ปิดขาย (1 = พร้อมขาย, 0 = สินค้าหมด/ปิดขาย)
);

-- สร้างตาราง Category
CREATE TABLE Category (
    Category_ID INTEGER PRIMARY KEY AUTOINCREMENT, -- รหัสหมวดหมู่
    Category_Name TEXT NOT NULL                    -- ชื่อหมวดหมู่ (เช่น น้ำซุป, วัตถุดิบสด, ของทานเล่น)
);

-- สร้างตาราง Employee
CREATE TABLE Employee (
    User_ID INTEGER PRIMARY KEY AUTOINCREMENT,    -- รหัสผู้ใช้งาน/พนักงาน
    Username TEXT UNIQUE,                         -- ชื่อผู้ใช้งานสำหรับเข้าสู่ระบบ
    Password TEXT,                                -- รหัสผ่าน
    Employee_Name TEXT,                           -- ชื่อ-นามสกุล พนักงาน
    Role TEXT                                     -- บทบาท/สิทธิ์: 'ผู้ดูแลระบบ/เจ้าของร้าน', 'พนักงานแคชเชียร์', 'พนักงานครัว', 'พนักงานเคาน์เตอร์'
);

-- สร้างตาราง Payment
CREATE TABLE Payment (
    Payment_ID INTEGER PRIMARY KEY AUTOINCREMENT, -- รหัสการชำระเงิน
    Order_ID INTEGER,                             -- รหัสออเดอร์ที่ชำระ (FK เชื่อมตาราง Order)
    Payment_Method TEXT,                          -- ช่องทางชำระเงิน: 'สแกน QR Code', 'เงินสด', 'บัตรเครดิต'
    Payment_Amount REAL,                          -- จำนวนเงินที่ลูกค้าจ่าย
    Change_Amount REAL,                           -- จำนวนเงินทอน (กรณีจ่ายเงินสด)
    Payment_Status TEXT,                          -- สถานะการชำระเงิน: 'รอชำระ', 'ชำระเงินเรียบร้อย', 'ล้มเหลว', 'ยกเลิก/คืนเงิน'
    Payment_Date_Time DATETIME DEFAULT CURRENT_TIMESTAMP, -- วันเวลาที่ชำระเงิน
    Transaction_Ref TEXT                          -- รหัสอ้างอิงสลิป/หมายเลขการทำรายการ
);

-- สร้างตาราง Order_Detail
CREATE TABLE Order_Detail (
    Order_Detail_ID INTEGER PRIMARY KEY AUTOINCREMENT, -- รหัสรายละเอียดรายการ
    Order_ID INTEGER,                             -- รหัสออเดอร์หลัก (FK เชื่อมตาราง Order)
    Menu_ID INTEGER,                              -- รหัสเมนูอาหาร (FK เชื่อมตาราง Menu)
    Quantity REAL,                                -- จำนวนที่สั่ง
    Unit_Price REAL,                              -- ราคาต่อหน่วย ณ ตอนสั่งซื้อ
    Subtotal REAL,                                -- ราคารวมของรายการนี้ (Quantity * Unit_Price)
    Spiciness_Level TEXT,                         -- ระดับความเผ็ด (เช่น เผ็ดน้อย, เผ็ดกลาง, เผ็ดมาก)
    Soup_Type TEXT,                               -- ประเภทน้ำซุปที่เลือก
    Special_Note TEXT                             -- หมายเหตุเพิ่มเติมจากลูกค้า (เช่น ไม่ใส่ผักชี)
);

-- สร้างตาราง Inventory
CREATE TABLE Inventory (
    Ingredient_ID INTEGER PRIMARY KEY AUTOINCREMENT, -- รหัสวัตถุดิบ
    Ingredient_Name TEXT,                         -- ชื่อวัตถุดิบ (เช่น พริกหม่าล่า, เนื้อสามชั้น)
    In_Stock_Quantity REAL,                       -- ปริมาณวัตถุดิบคงเหลือในคลัง
    Unit TEXT,                                    -- หน่วยนับ (เช่น กิโลกรัม, กรัม, ชิ้น)
    Reorder_Level REAL                            -- ปริมาณขั้นต่ำสำหรับแจ้งเตือนให้สั่งวัตถุดิบเพิ่ม
);
